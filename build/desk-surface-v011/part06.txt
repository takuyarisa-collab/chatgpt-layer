else setContentEditableValue(editor, text);

async function setEditorText(editor, text) {
var expected = normalizeEditorText(text);
writeEditorText(editor, text);
await waitForEditorCommit();
var current = findEditor();
if (current && current !== editor && getEditorText(current) !== expected) {
writeEditorText(current, text);
await waitForEditorCommit();
}
current = findEditor() || editor;
var actual = getEditorText(current);
return {
ok: isEditorInteractive(current) && actual === expected,
editor: current,
actual: actual
};
}

function normalizeAction(raw) {
if (!isRecord(raw)) return null;
if (typeof raw.id !== "string" || !/^[a-z0-9_-]{1,48}$/i.test(raw.id)) return null;
if (raw.enabled === false) return null;
var type = typeof raw.type === "string" ? raw.type : "";
if (["insertPrompt", "reload", "scrollToBottom"].indexOf(type) < 0) return null;
var label = typeof raw.label === "string" ? raw.label.trim().slice(0, 8) : "";
if (!label) return null;
var action = {
id: raw.id,
label: label,
title: typeof raw.title === "string" && raw.title.trim() ? raw.title.trim().slice(0, 100) : label,
type: type,
order: clampNumber(raw.order, 0, -1000, 1000),
style: normalizeButtonStyle(raw.style, DEFAULT_LAYER_BUTTON_STYLE)
};
if (type === "insertPrompt") {
if (typeof raw.text !== "string" || !raw.text.length) return null;
action.text = raw.text.slice(0, 4000);
action.mode = ["empty", "append", "replace"].indexOf(raw.mode) >= 0 ? raw.mode : "empty";
action.separator = typeof raw.separator === "string" ? raw.separator.slice(0, 20) : "\n";
action.confirmReplace = raw.confirmReplace !== false;
}
return action;
}
function normalizeActions(value) {
if (!Array.isArray(value)) return [];
var actions = [];
var seen = {};
value.forEach(function (raw) {
var action = normalizeAction(raw);
if (!action || seen[action.id]) return;
seen[action.id] = true;
actions.push(action);
});
actions.sort(function (left, right) {
if (left.order !== right.order) return left.order - right.order;
return left.id.localeCompare(right.id);
});
return actions.slice(0, 8);
}

async function runInsertPromptAction(action) {
var editor = findEditor();
if (!editor) {
showToast("表示中の入力欄が見つかりませんでした。", true);
return;
}
var existing = getEditorText(editor);
var nextText = action.text;
if (action.mode === "empty" && existing) {
showToast("入力中の文章があるため、挿入しませんでした。", true);
return;
}
if (action.mode === "append" && existing) nextText = existing + action.separator + action.text;
if (action.mode === "replace" && existing && action.confirmReplace && !window.confirm("入力中の文章を置き換えますか？")) return;
try {
var result = await setEditorText(editor, nextText);
if (!result.ok) {
console.warn("[ChatGPT Layer] insertPrompt verification failed", {
expected: normalizeEditorText(nextText),
actual: result.actual,
editor: result.editor
});
showToast("入力欄への反映を確認できませんでした。", true);
return;
}
showToast("入力欄へ「" + action.label + "」をセットしました。", false);
} catch (error) {
console.warn("[ChatGPT Layer] insertPrompt failed", error);
showToast("入力欄への挿入に失敗しました。", true);
}
}
