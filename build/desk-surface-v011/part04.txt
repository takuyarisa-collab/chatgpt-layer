var viewport = getVisualViewportBounds();
var right = viewport.left + viewport.width;
var bottom = viewport.top + viewport.height;
return rect.right > viewport.left && rect.left < right && rect.bottom > viewport.top && rect.top < bottom;
}
function editorHitTestScore(editor) {
var rect = editor.getBoundingClientRect();
var x = Math.min(rect.right - 2, Math.max(rect.left + 2, rect.left + rect.width / 2));
var y = Math.min(rect.bottom - 2, Math.max(rect.top + 2, rect.top + rect.height / 2));
var hit = document.elementFromPoint(x, y);
if (!hit) return 0;
if (hit === editor || editor.contains(hit)) return 500;
if (hit instanceof Element && hit.contains(editor)) return 160;
return 0;
}
function scoreEditor(editor, domIndex) {
var score = domIndex / 1000;
var rect = editor.getBoundingClientRect();
var viewport = getVisualViewportBounds();
var viewportBottom = viewport.top + viewport.height;
var active = document.activeElement;
if (active === editor || editor.contains(active)) score += 1200;
if (editor.id === "prompt-textarea") score += 420;
if (editor.closest('[data-type="unified-composer"], [data-testid="composer"], [data-testid="composer-surface"]')) score += 360;
if (editor.closest("form")) score += 120;
if (editor.isContentEditable) score += 80;
score += editorHitTestScore(editor);
score += Math.max(0, 280 - Math.abs(viewportBottom - rect.bottom));
score += Math.min(100, rect.width * rect.height / 1600);
return score;
}
function findEditorCandidates() {
var candidates = Array.from(document.querySelectorAll(EDITOR_SELECTOR));
var active = document.activeElement;
if (active instanceof Element) {
var activeEditor = active.matches(EDITOR_SELECTOR) ? active : active.closest(EDITOR_SELECTOR);
if (activeEditor && candidates.indexOf(activeEditor) < 0) candidates.push(activeEditor);
}
return candidates;
}
function findEditor() {
return findEditorCandidates()
.map(function (editor, index) {
return {
editor: editor,
score: isEditorInteractive(editor) ? scoreEditor(editor, index) : -Infinity
};
})
.filter(function (candidate) { return Number.isFinite(candidate.score); })
.sort(function (left, right) { return right.score - left.score; })
.map(function (candidate) { return candidate.editor; })[0] || null;
}
function parseRadius(value) {
var number = parseFloat(value);
return Number.isFinite(number) ? number : 0;
}
function isDeskElement(element) {
return element && (
element.hasAttribute(DESK_SURFACE_ATTRIBUTE) ||
element.hasAttribute(DESK_MAT_ATTRIBUTE)
);
}
function findComposerSurface() {
var editor = findEditor();
if (!editor) return null;
var explicit = editor.closest('[data-type="unified-composer"], [data-testid="composer"], [data-testid="composer-surface"]');
if (explicit && !isDeskElement(explicit)) return explicit;
var form = editor.closest("form");
var editorRect = editor.getBoundingClientRect();
var element = editor.parentElement;
while (element && element !== document.body) {
if (element instanceof HTMLElement && !isDeskElement(element)) {
var rect = element.getBoundingClientRect();
var computed = getComputedStyle(element);
var radius = Math.max(
parseRadius(computed.borderTopLeftRadius),
parseRadius(computed.borderTopRightRadius),
parseRadius(computed.borderBottomLeftRadius),
parseRadius(computed.borderBottomRightRadius)
);
if (
radius >= 10 &&
rect.width >= Math.max(220, editorRect.width) &&
rect.height >= editorRect.height &&
rect.height <= 320
) return element;
}
if (element === form) break;
element = element.parentElement;
}
return form || editor.parentElement;
}
function rememberComposerInlineStyles(composer) {
if (composerOriginalStyles.has(composer)) return;
var snapshot = {};
COMPOSER_INLINE_PROPERTIES.forEach(function (property) {
snapshot[property] = {
value: composer.style.getPropertyValue(property),
priority: composer.style.getPropertyPriority(property)
};
});
composerOriginalStyles.set(composer, snapshot);
}
function restoreComposerInlineStyles(composer) {
var snapshot = composerOriginalStyles.get(composer);
if (!snapshot) return;
COMPOSER_INLINE_PROPERTIES.forEach(function (property) {
var entry = snapshot[property];
if (entry.value) composer.style.setProperty(property, entry.value, entry.priority || "");
else composer.style.removeProperty(property);
});
composerOriginalStyles.delete(composer);
}
function removeDeskStructure(surface, restoreComposer) {
if (!(surface instanceof HTMLElement)) return;
var mat = surface.querySelector('[' + DESK_MAT_ATTRIBUTE + '="on"]');
var composer = mat && mat.querySelector(':scope > [' + COMPOSER_SLOT_ATTRIBUTE + '="on"]');
if (composer instanceof HTMLElement) {
restoreComposerInlineStyles(composer);
composer.removeAttribute(COMPOSER_SLOT_ATTRIBUTE);
composer.removeAttribute(COMPOSER_SURFACE_ATTRIBUTE);
if (restoreComposer && surface.parentNode && composer.parentNode === mat) {
surface.parentNode.insertBefore(composer, surface);
}
}
surface.remove();
}
function clearDeskStructures(restoreComposer, keepComposer) {
document.querySelectorAll('[' + DESK_SURFACE_ATTRIBUTE + ']').forEach(function (surface) {
var composer = surface.querySelector('[' + COMPOSER_SLOT_ATTRIBUTE + '="on"]');
if (keepComposer && composer === keepComposer) return;
removeDeskStructure(surface, restoreComposer && !keepComposer);
});
if (!keepComposer) {
document.querySelectorAll('[' + COMPOSER_SLOT_ATTRIBUTE + ']').forEach(function (composer) {
restoreComposerInlineStyles(composer);
composer.removeAttribute(COMPOSER_SLOT_ATTRIBUTE);
});
if (deskResizeObserver) deskResizeObserver.disconnect();
}
}
function observeDeskStructure(surface, mat, composer) {
if (!deskResizeObserver) return;
deskResizeObserver.disconnect();
[surface, mat, composer].forEach(function (element) {
if (element instanceof Element) deskResizeObserver.observe(element);
});
}
function ensureDeskStructure(composer) {
if (!(composer instanceof HTMLElement) || !composer.parentNode) return null;
var existingMat = composer.parentElement;
var existingSurface = existingMat && existingMat.parentElement;
var validExisting = existingMat &&
existingMat.getAttribute(DESK_MAT_ATTRIBUTE) === "on" &&
existingSurface &&
existingSurface.getAttribute(DESK_SURFACE_ATTRIBUTE) === "on";
clearDeskStructures(false, validExisting ? composer : null);
if (validExisting) {
composer.setAttribute(COMPOSER_SLOT_ATTRIBUTE, "on");
composer.setAttribute(COMPOSER_SURFACE_ATTRIBUTE, "desk");
observeDeskStructure(existingSurface, existingMat, composer);
return { surface: existingSurface, mat: existingMat, composer: composer };
}
var parent = composer.parentNode;
var surface = document.createElement("div");
var mat = document.createElement("div");
surface.setAttribute(DESK_SURFACE_ATTRIBUTE, "on");
mat.setAttribute(DESK_MAT_ATTRIBUTE, "on");
rememberComposerInlineStyles(composer);
parent.insertBefore(surface, composer);
surface.appendChild(mat);
mat.appendChild(composer);
composer.setAttribute(COMPOSER_SLOT_ATTRIBUTE, "on");
composer.setAttribute(COMPOSER_SURFACE_ATTRIBUTE, "desk");
composer.style.setProperty("width", "100%", "important");
composer.style.setProperty("max-width", "100%", "important");
composer.style.setProperty("min-width", "0", "important");
composer.style.setProperty("margin", "0", "important");
composer.style.setProperty("box-sizing", "border-box", "important");
observeDeskStructure(surface, mat, composer);
return { surface: surface, mat: mat, composer: composer };
}
function clearComposerDecoration() {
clearDeskStructures(true, null);
document.querySelectorAll('[' + COMPOSER_SURFACE_ATTRIBUTE + ']').forEach(function (element) {
element.removeAttribute(COMPOSER_SURFACE_ATTRIBUTE);
});
}
function applyComposerDecoration(theme) {
var deskEnabled = Boolean(theme && theme.deskSurface && theme.deskMat);
var standardEnabled = Boolean(theme && theme.composerStyle && !deskEnabled);
var current = deskEnabled || standardEnabled ? findComposerSurface() : null;
document.querySelectorAll('[' + COMPOSER_SURFACE_ATTRIBUTE + ']').forEach(function (element) {
